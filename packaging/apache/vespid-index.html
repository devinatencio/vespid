Vespid Packages

RPM and DEB packages for Vespid.

FormatPackages
RPM4
DEB4

RPM

sudo curl -fsSL https://packages.vespid.app/keys/vespid-packages.asc -o /etc/pki/rpm-gpg/RPM-GPG-KEY-vespid
sudo tee /etc/yum.repos.d/vespid.repo <<'REPO'
[vespid]
name=Vespid Packages
baseurl=https://packages.vespid.app/rpm/$basearch
enabled=1
gpgcheck=1
repo_gpgcheck=1
gpgkey=file:///etc/pki/rpm-gpg/RPM-GPG-KEY-vespid
REPO
sudo dnf install vespid vespid-agent vespid-server vespid-sync

DEB

curl -fsSL https://packages.vespid.app/keys/vespid-packages.asc | sudo gpg --dearmor -o /usr/share/keyrings/vespid-packages.gpg
echo 'deb [signed-by=/usr/share/keyrings/vespid-packages.gpg] https://packages.vespid.app/deb stable main' | sudo tee /etc/apt/sources.list.d/vespid.list
sudo apt update
sudo apt install vespid vespid-agent vespid-server vespid-sync
